Day 6 — Ratings & Reviews System

(Trust, data integrity & real-world logic)

🎯 Day 6 Goal

By the end of today, you will have:

⭐ Ratings & reviews for workers

🔒 Only valid clients can review

🚫 No fake reviews

📊 Worker average rating

🧠 System design: data integrity

🧠 System Design (Beginner, Clear)
❓ Real-world rule

A client should be able to review a worker only if a job was completed together.

So we enforce:

Client → Job → Assigned Worker → Review


📌 Key design principle

“Reviews must be tied to completed jobs to prevent fake ratings.”

1️⃣ Review Model
Create file

backend/src/models/Review.js

const mongoose = require("mongoose");

const reviewSchema = new mongoose.Schema(
  {
    jobId: {
      type: mongoose.Schema.Types.ObjectId,
      ref: "Job",
      required: true
    },
    workerId: {
      type: mongoose.Schema.Types.ObjectId,
      ref: "User",
      required: true
    },
    clientId: {
      type: mongoose.Schema.Types.ObjectId,
      ref: "User",
      required: true
    },
    rating: {
      type: Number,
      min: 1,
      max: 5,
      required: true
    },
    comment: {
      type: String
    }
  },
  { timestamps: true }
);

module.exports = mongoose.model("Review", reviewSchema);

2️⃣ Review Controller
Create file

backend/src/controllers/reviewController.js

const Review = require("../models/Review");
const Job = require("../models/Job");
const WorkerProfile = require("../models/WorkerProfile");

exports.addReview = async (req, res) => {
  try {
    const { rating, comment } = req.body;
    const jobId = req.params.jobId;

    // 1. Check job exists & completed
    const job = await Job.findById(jobId);
    if (!job || job.status !== "completed") {
      return res.status(400).json({ message: "Job not completed" });
    }

    // 2. Prevent duplicate review
    const alreadyReviewed = await Review.findOne({
      jobId,
      clientId: req.user.id
    });

    if (alreadyReviewed) {
      return res.status(400).json({ message: "Review already submitted" });
    }

    // 3. Create review
    const review = await Review.create({
      jobId,
      workerId: job.workerId,
      clientId: req.user.id,
      rating,
      comment
    });

    // 4. Update worker average rating
    const reviews = await Review.find({ workerId: job.workerId });
    const avg =
      reviews.reduce((sum, r) => sum + r.rating, 0) / reviews.length;

    await WorkerProfile.findOneAndUpdate(
      { userId: job.workerId },
      { ratingAvg: avg }
    );

    res.status(201).json(review);
  } catch (error) {
    res.status(500).json({ message: "Error adding review" });
  }
};

3️⃣ Review Routes
Create file

backend/src/routes/reviewRoutes.js

const express = require("express");
const { addReview } = require("../controllers/reviewController");
const { protect } = require("../middlewares/authMiddleware");
const { authorize } = require("../middlewares/roleMiddleware");

const router = express.Router();

// Client adds review after job completion
router.post(
  "/:jobId",
  protect,
  authorize("client"),
  addReview
);

module.exports = router;

4️⃣ Register Review Routes
In app.js
const reviewRoutes = require("./routes/reviewRoutes");
app.use("/api/reviews", reviewRoutes);
