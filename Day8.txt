2️⃣ Chat Controller (Create / Fetch)
backend/src/controllers/chatController.js
const ChatRoom = require("../models/ChatRoom");
const Message = require("../models/Message");

exports.createOrGetChatRoom = async (req, res) => {
  const { jobId, otherUserId } = req.body;

  let room = await ChatRoom.findOne({ jobId });

  if (!room) {
    room = await ChatRoom.create({
      jobId,
      participants: [req.user.id, otherUserId]
    });
  }

  res.json(room);
};

exports.getMessages = async (req, res) => {
  const messages = await Message.find({
    chatRoomId: req.params.roomId
  }).sort({ createdAt: 1 });

  res.json(messages);
};

3️⃣ Chat Routes

4️⃣ Socket.IO Chat Events
Update server.js (inside io.on("connection"))
const Message = require("./src/models/Message");

io.on("connection", (socket) => {
  socket.on("joinRoom", (roomId) => {
    socket.join(roomId);
  });

  socket.on("sendMessage", async (data) => {
    const message = await Message.create({
      chatRoomId: data.roomId,
      senderId: data.senderId,
      text: data.text
    });

    io.to(data.roomId).emit("receiveMessage", message);
  });
});

✅ Solution

Use Redis Pub/Sub

Socket Server 2
Socket Server 3
       ↓
     Redis


📌 Redis synchronizes socket events.
